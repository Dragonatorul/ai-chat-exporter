type Payload = { id: string; value: number };

function validatePayload(payload: Payload): boolean {
  return !!payload.id && Number.isFinite(payload.value);
}

export function handlePayload(payload: Payload): string {
  if (!validatePayload(payload)) {
    throw new Error("Invalid payload");
  }
  return `processed:${payload.id}:${payload.value}`;
}
